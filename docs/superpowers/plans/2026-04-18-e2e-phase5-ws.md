```python
"""Minimal async WebSocket client for E2E tests.

Pattern:
    wc = await WsClient.connect(f"ws://web:8000/ws/threads/{tid}/")
    # trigger backend...
    await wc.wait_for_event("message_done", timeout=10.0)
    wc.assert_sequence(["message_started", "text_delta", "message_done"])
    await wc.close()
"""
from __future__ import annotations

import asyncio
import json
from typing import Any


class WsClient:
    def __init__(self, conn: Any) -> None:
        self.conn = conn
        self._events: list[dict] = []
        self._event_queue: asyncio.Queue[dict] = asyncio.Queue()
        self._task: asyncio.Task | None = None

    @classmethod
    async def connect(cls, url: str) -> "WsClient":
        import websockets
        conn = await websockets.connect(url, open_timeout=5, close_timeout=5)
        wc = cls(conn)
        await wc.start()
        return wc

    async def start(self) -> None:
        self._task = asyncio.create_task(self._reader())

    async def _reader(self) -> None:
        try:
            while True:
                payload = await self.conn.recv()
                try:
                    event = json.loads(payload)
                except Exception:
                    event = {"type": "unparseable", "raw": payload}
                self._events.append(event)
                await self._event_queue.put(event)
        except asyncio.CancelledError:
            raise
        except Exception:
            # Connection closed
            pass

    async def wait_for_event(self, type_: str, timeout: float = 10.0) -> dict:
        deadline = asyncio.get_event_loop().time() + timeout
        # First check already-received events
        for e in self._events:
            if e.get("type") == type_:
                return e
        while True:
            remaining = deadline - asyncio.get_event_loop().time()
            if remaining <= 0:
                raise TimeoutError(f"did not receive event of type {type_} within {timeout}s")
            e = await asyncio.wait_for(self._event_queue.get(), timeout=remaining)
            if e.get("type") == type_:
                return e

    def events_of(self, type_: str) -> list[dict]:
        return [e for e in self._events if e.get("type") == type_]

    def assert_sequence(self, expected_types: list[str]) -> None:
        """Assert event types appear in the given order (subset match — other events may interleave)."""
        seen = iter(e.get("type") for e in self._events)
        for t in expected_types:
            for actual in seen:
                if actual == t:
                    break
            else:
                raise AssertionError(f"expected type {t} not found after previous matches; got {[e.get('type') for e in self._events]}")

    async def send_json(self, payload: dict) -> None:
        await self.conn.send(json.dumps(payload))

    async def close(self) -> None:
        if self._task:
            self._task.cancel()
            try:
                await self._task
            except asyncio.CancelledError:
                pass
        await self.conn.close()
```

```python
"""Reconnect within 5s replays recent events — no gap."""
from __future__ import annotations

import asyncio

import pytest
import httpx

from e2e.helpers.ws_client import WsClient


@pytest.mark.asyncio
@pytest.mark.integration
async def test_ws_reconnect_replays_recent_events(ws_base_url, api_base_url, threads) -> None:
    from apps.threads.models import Thread
    t = Thread.objects.get(title="E2E plain thread")

    wc1 = await WsClient.connect(f"{ws_base_url}/ws/threads/{t.id}/")
    httpx.post(f"{api_base_url}/api/threads/{t.id}/messages/", json={"text": "hi"}, timeout=5)

    # Wait for at least one delta, then close
    await wc1.wait_for_event("text_delta", timeout=10.0)
    last_seq = wc1._events[-1].get("data", {}).get("seq", 0)
    await wc1.close()

    await asyncio.sleep(1)

    # Reconnect with last-seq; expect no gap
    wc2 = await WsClient.connect(f"{ws_base_url}/ws/threads/{t.id}/?since={last_seq}")
    try:
        await wc2.wait_for_event("message_done", timeout=15.0)
        seqs = [e.get("data", {}).get("seq") for e in wc2._events if "seq" in e.get("data", {})]
        # Ensure contiguous
        for a, b in zip(seqs, seqs[1:]):
            assert b == a + 1
    finally:
        await wc2.close()
```
